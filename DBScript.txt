use urban_waste_db

// Query #2: Find high-risk incidents that are not contained.
db.hazardousIncidents.find(
  {
    hazard_level: { $gte: 4.0 },
    is_contained: false
  },
  {
    _id: 0,
    incident_id: 1,
    hazard_level: 1,
    is_contained: 1,
    status: 1
  }
).sort({ hazard_level: -1 })

// Query #3: Find incidents involving selected waste types.
db.hazardousIncidents.find(
  {
    waste_type: {
      $in: ["asbestos", "industrial_solvents"]
    }
  },
  {
    _id: 0,
    incident_id: 1,
    waste_type: 1,
    hazard_level: 1
  }
)

// Query #4: Find confirmed violations with unpaid fines.
db.patrolInspections.find(
  {
    infraction_confirmed: true,
    fine_issued: true,
    "fine_details.payment_status": {
      $exists: true,
      $not: /^paid$/i
    }
  },
  {
    _id: 0,
    inspection_id: 1,
    incident_ref: 1,
    violations_detected: 1,
    fine_details: 1
  }
)

// Query #5: Find the five districts with the most unresolved incidents.
db.hazardousIncidents.aggregate([
  {
    $match: {
      status: { $nin: ["Remediated", "Closed"] }
    }
  },
  {
    $group: {
      _id: "$address.district",
      unresolved_incidents: { $sum: 1 }
    }
  },
  {
    $sort: { unresolved_incidents: -1 }
  },
  {
    $limit: 5
  }
])

// Query #6: Find the most active verified reporters in the last six months.
db.hazardousIncidents.aggregate([
  {
    $match: {
      "citizen_reporter.verified_reporter": true,
      reported_date: {
        $gte: new Date(new Date().setMonth(new Date().getMonth() - 6))
      }
    }
  },
  {
    $group: {
      _id: "$citizen_reporter.username",
      reports: { $sum: 1 }
    }
  },
  {
    $sort: { reports: -1 }
  }
])

// Query #10: Find incidents reported within a date range.
db.hazardousIncidents.find(
  {
    reported_date: {
      $gte: ISODate("2026-01-01T00:00:00Z"),
      $lt: ISODate("2026-04-01T00:00:00Z")
    }
  },
  {
    _id: 0,
    incident_id: 1,
    reported_date: 1,
    status: 1,
    address: 1
  }
).sort({ reported_date: 1 })
